- 添加调试日志便于追踪

```python
# 处理段落占位符（送达材料清单的paragraph类型占位符）
try:
    from template_manifest import get_paragraph_fills
    from manifest_word_fill import _paragraph_range, _table_range_start

    paragraph_fills = get_paragraph_fills(manifest)
    if paragraph_fills:
        table_start = _table_range_start(doc)
        for p in paragraph_fills:
            idx = p.get("index")
            if not idx:
                continue
            ph = p.get("placeholder")
            if not ph:
                continue
            val = field_patch.get(ph, "" if p.get("role") == "clear" else None)
            if val is None and p.get("role") != "clear":
                continue

            # 调试日志
            if template_name == "送达材料清单" and "律师" in ph:
                print(f"  [DEBUG textbox paragraph] 尝试填充段落占位符: {ph} = \"{val}\"")

            try:
                rng = _paragraph_range(doc, idx, table_start)
                off = {ph: p["offset"]} if p.get("offset") else {}
                n = replace_placeholders_atomic(
                    doc,
                    rng,
                    {ph: str(val)},
                    blacken_fn=blacken_fn,
                    placeholder_offsets=off,
                )
                if n:
                    total += n
                    # 调试日志
                    if template_name == "送达材料清单" and "律师" in ph:
                        print(f"  [DEBUG textbox paragraph] 成功填充段落占位符: {ph}")
            except Exception as e:
                import sys
                print(f"  [WARN] 文本框段落填充 idx={idx}, ph={ph}: {e}", file=sys.stderr)
except Exception as e:
    import sys
    print(f"  [WARN] 处理段落占位符时出错: {e}", file=sys.stderr)
```

```python
def _deduplicate_units_by_catalog_seq(units: List[DocumentUnit], log=print) -> List[DocumentUnit]:
    # 按catalog_seq分组
    seq_groups = defaultdict(list)
    for unit in units:
        if unit.catalog_seq is not None:
            seq_groups[unit.catalog_seq].append(unit)
    
    # 对每组进行合并
    for seq, group in seq_groups.items():
        if len(group) > 1:
            # 选择置信度最高的作为基础
            best_unit = max(group, key=lambda u: u.confidence or 0)
            # 合并页码范围
            min_page = min(u.start_page for u in group)
            max_page = max(u.end_page for u in group)
            # 创建合并后的DocumentUnit
            merged_unit = DocumentUnit(
                doc_id=best_unit.doc_id,
                doc_type=best_unit.doc_type,
                start_page=min_page,
                end_page=max_page,
                title=best_unit.title,
                catalog_seq=seq,
                source_path=best_unit.source_path,
                score=best_unit.score,
                confidence=best_unit.confidence,
            )
```
